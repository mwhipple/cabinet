#+BEGIN_SRC dockerfile :tangle "Dockerfile"
FROM debian:bookworm-slim

# Install coreutils and emacs
RUN apt-get update && apt-get install -y \
    coreutils \
    emacs-nox \
    python3 \
    file \
    ghc \
    git \
    pandoc \
    && rm -rf /var/lib/apt/lists/*

# Create a non-root user for Emacs
RUN useradd -m -s /bin/bash al

# Set up environment variables
ENV HOME=/home/al
WORKDIR $HOME

USER al
